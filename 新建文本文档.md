class PageResult<T> {
  final List<T> itemList;
  final bool isLoading;
  final bool hasMore;
  final int? lastId;
  final String? error;

  const PageResult({
    this.itemList = const [],
    this.isLoading = false,
    this.hasMore = true,
    this.lastId,
    this.error,
  });

  PageResult<T> copyWith({
    List<T>? itemList,
    bool? isLoading,
    bool? hasMore,
    String? error,
    int? lastId,
  }) {
    //如果传了新值用新值，否则用旧值
    return PageResult(
      itemList: itemList ?? this.itemList,
      isLoading: isLoading ?? this.isLoading,
      hasMore: hasMore ?? this.hasMore,
      lastId: lastId ?? this.lastId,
      error: error,
    );
  }
}

class PageFetchController<T> {

  /// 业务方提供的请求函数
  /// 入参：当前请求参数
  /// 返回：服务端响应
  final Future<PageResponse<T>> Function(PageParam param) onFetch;

  PageFetchController({
    required this.onFetch, // 业务方只需实现这一个函数
    int pageSize = 20,
  }) : _param = PageParam(pageSize: pageSize);



  /// 当前 UI 状态
  PageResult<T> _result = const PageResult();
  PageResult<T> get result => _result;

  /// 当前请求参数（内部维护，外部无需关心）
  PageParam _param;

  /// 下拉刷新
  Future<void> refresh() => _fetch(isRefresh: true);

  /// 上拉加载更多
  Future<void> loadMore() => _fetch(isRefresh: false);

  /// 核心分页逻辑
  Future<void> _fetch({required bool isRefresh}) async {
    // 防止重复请求
    if (_result.isLoading) return;

    // 没有更多数据时，不发起加载更多请求
    if (!isRefresh && !_result.hasMore) return;

    // 更新加载状态
    _result = _result.copyWith(isLoading: true, error: null);
    _notify();

    // 更新游标参数
    _param = isRefresh
        ? _param.toRefresh()
        : _param.toNextPage(_result.lastId);

    try {
      final response = await onFetch(_param);

      _result = PageResult(
        itemList: isRefresh
            ? response.list
            : [..._result.itemList, ...response.list],
        isLoading: false,
        hasMore: response.hasMore,
        lastId: response.lastId,
      );
    } catch (e) {
      _result = _result.copyWith(
        isLoading: false,
        error: e.toString(),
      );
    }

    _notify();
  }

  // ---------- 状态通知（根据你的状态管理方案选择一种）----------

  /// 方案A：回调通知（无任何框架依赖）
  VoidCallback? onStateChanged;
  void _notify() => onStateChanged?.call();

  /// 方案B：如果用 GetX，把 _result 换成 Rx<PageResult<T>>
  /// 方案C：如果用 Riverpod，配合 StateNotifier 使用
}

class PageFetchController<T> extends ChangeNotifier {
  // 1. 核心状态：暴露给 UI 层
  PageResult<T> value = const PageResult<T>();

  // 2. 数据获取函数：由业务侧在实例化时注入
  final PageFetcher<T> fetcher;

  // 3. 内部参数状态
  PageParam _currentParam;

  PageFetchController({
    required this.fetcher,
    int initialPageSize = 20,
  }) : _currentParam = PageParam(pageSize: initialPageSize);

  /// 初始化并加载首屏数据
  void initFetch() {
    refresh();
  }

  /// 下拉刷新
  Future<void> refresh() => _fetchData(isRefresh: true);

  /// 上拉加载更多
  Future<void> loadMore() => _fetchData(isRefresh: false);

  /// 核心加载逻辑
  Future<void> _fetchData({required bool isRefresh}) async {
    // 1. 防抖与拦截
    if (value.isLoading) return; 
    if (!isRefresh && !value.hasMore) return; // 不是刷新且没有更多数据时拦截

    // 2. 更新状态为加载中，清空之前的错误
    value = value.copyWith(isLoading: true, error: null);
    notifyListeners();

    // 3. 准备本次请求的参数
    if (isRefresh) {
      _currentParam = _currentParam.toRefresh();
    } else {
      _currentParam = _currentParam.toNextPage(value.lastId);
    }

    try {
      // 4. 调用外部注入的 fetcher 真正发起请求
      final PageResponse<T> response = await fetcher(_currentParam);

      // 5. 合并新老数据
      final List<T> newItems = isRefresh 
          ? response.list 
          : [...value.itemList, ...response.list];

      // 6. 更新成功状态
      value = PageResult(
        itemList: newItems,
        isLoading: false,
        hasMore: response.hasMore,
        lastId: response.lastId,
        error: null,
      );
    } catch (e) {
      // 7. 更新失败状态
      value = value.copyWith(
        isLoading: false, 
        error: e.toString(),
      );
    } finally {
      notifyListeners();
    }
  }
}
